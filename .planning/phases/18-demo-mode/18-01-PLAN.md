```typescript
// client/src/lib/demoSession.ts — REQUIRED EXPORTS
export const DEMO_ROOM_ORDER = [
  'reception',
  'er',
  'break_room',
  'records_room',
] as const;

export type DemoRoomId = typeof DEMO_ROOM_ORDER[number];

export interface DemoSessionState {
  active: boolean;
  startedAt: number;
  completedRooms: DemoRoomId[];
}

export function startDemo(): void;             // sets active=true, resets completedRooms
export function endDemo(): void;               // sets active=false, clears state
export function isDemoActive(): boolean;
export function getDemoRoomOrder(): readonly DemoRoomId[];
export function getDemoRoomIndex(currentRoomId: string): number;  // -1 if not in demo order
export function markRoomComplete(roomId: string): void;           // no-op if not a demo room
export function getCompletedDemoRooms(): DemoRoomId[];
```

- `endDemo()` must clear state so the next `startDemo()` is fresh.
