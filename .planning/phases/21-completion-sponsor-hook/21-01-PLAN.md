### NEW: `client/src/data/spriteAssetPaths.ts`

A small map keyed off the same Phaser texture keys BootScene preloads. The CertificateOverlay reads from this map to render the NPC sprite as a React `<img>` (we cannot share Phaser textures into React without canvas extraction; the simplest path is a parallel asset path lookup mirroring BootScene).

```ts
/**
 * spriteAssetPaths.ts — React-side mirror of BootScene NPC spritesheet preload paths.
 *
 * BootScene loads these spritesheets for use in Phaser scenes; the Phase 21 sponsor
 * certificate overlay needs to render the configured NPC sprite as a plain <img>
 * element (React overlay, no Phaser canvas), so we expose the source paths through
 * this small parallel map. Keep in sync with BootScene.preload() if NPC sheets are
 * added/renamed.
 *
 * The keys match SPONSOR_CONFIG.character_sprite values (e.g., 'npc_staff_sheet').
 */
import.meta.env  // ensure Vite picks up BASE_URL handling

const base = import.meta.env.BASE_URL;

export const SPONSOR_SPRITE_PATHS: Record<string, string> = {
  npc_receptionist_sheet: `${base}attached_assets/generated_images/privacyquest/characters/npc_receptionist.png`,
  npc_nurse_sheet:        `${base}attached_assets/generated_images/privacyquest/characters/npc_nurse.png`,
  npc_doctor_sheet:       `${base}attached_assets/generated_images/privacyquest/characters/npc_doctor.png`,
  npc_it_tech_sheet:      `${base}attached_assets/generated_images/privacyquest/characters/npc_it_tech.png`,
  npc_officer_sheet:      `${base}attached_assets/generated_images/privacyquest/characters/npc_officer.png`,
  npc_boss_sheet:         `${base}attached_assets/generated_images/privacyquest/characters/npc_boss.png`,
  npc_staff_sheet:        `${base}attached_assets/generated_images/privacyquest/characters/npc_staff.png`,
  npc_patient_sheet:      `${base}attached_assets/generated_images/privacyquest/characters/npc_patient.png`,
  npc_visitor_sheet:      `${base}attached_assets/generated_images/privacyquest/characters/npc_visitor.png`,
};

/** Resolve a sponsor character_sprite key to its source path; falls back to staff if unknown. */
export function getSponsorSpritePath(spriteKey: string): string {
  return SPONSOR_SPRITE_PATHS[spriteKey] ?? SPONSOR_SPRITE_PATHS.npc_staff_sheet;
}
```

### NEW: `client/src/components/CertificateOverlay.tsx`

The whole capstone overlay. State machine drives the sequence; click/keyboard advances dialogue; copy button uses clipboard API with fallback.

**Phase machine:**
```
mount: phase = 'dim'
  setTimeout 400ms → phase = 'beat'
  setTimeout 500ms → phase = 'fanfare'  (emit sfx_fanfare here)
  setTimeout 600ms → phase = 'npc'      (NPC + line 1 appear)
  click/space → phase = 'line2'
  click/space → phase = 'cert'          (sponsor name + code + COPY + RETURN buttons)
```

**Key implementation details:**

1. **Audio:** emit `REACT_PLAY_SFX` with `sfx_fanfare` (volume 0.9) on entering `'fanfare'` phase. Emit `sfx_interact` on dialogue advance and copy success.

2. **Sprite rendering:** Use a 32×32 `<div>` with `background-image: url(...)` from `getSponsorSpritePath(SPONSOR_CONFIG.character_sprite)`. Spritesheet is 96×128 (3 cols × 4 rows), so frame 0 (idle-down, top-left) renders cleanly with `background-position: 0 0` and `background-size: 96px 128px`. Scale up the container to 128×128 with `image-rendering: pixelated` for crisp 4× upscale.

3. **Dialogue rendering:** Two-line dialogue appears in a JRPG-style speech box BELOW the NPC. Line 1 first; click/Space → line 1 stays + line 2 appears underneath; click/Space → certificate body reveals (NPC and dialogue stay on the side; cert card animates in beside them).

4. **Copy button:** `navigator.clipboard.writeText(SPONSOR_CONFIG.code)`. On success: button flashes (color shift 200ms), label changes to "COPIED ✓" for 1200ms, plays `sfx_sorter_correct` (the chime — already preloaded). Fallback if clipboard API unavailable: hidden `<textarea>` + `document.execCommand('copy')`.

5. **Return-to-menu button:** `onReturn` callback prop → `UnifiedGamePage` calls `endDemo()` then `window.location.reload()`.

6. **Keyboard:** `keydown` listener for Space/Enter → advances dialogue or, if on `'cert'`, focuses the COPY button. Esc returns to menu directly.

7. **Z-index:** `z-[200]` (above the existing `z-[150]` banner).

**Sketch (full file with sequence machine + certificate body + dialogue + copy button — author has discretion on exact CSS values):**

```tsx
/**
 * CertificateOverlay — Phase 21 capstone: completion sequence + sponsor handoff.
 *
 * CERT-01: ordered sequence dim → ~500ms beat → fanfare → certificate animation
 *          → sponsor code reveal (Commandment 2: anticipation before reward).
 * CERT-02: monospace code box + copy-to-clipboard with audio + visual confirm.
 * CERT-03: end NPC handoff — sprite + two dialogue lines from SPONSOR_CONFIG.
 *
 * Demo-only path (rendered only when UnifiedGamePage's pageMode === 'demo-complete').
 * The full-game flow has its own EndScreen and is untouched.
 */

import { useEffect, useState, useCallback, useRef } from 'react';
import { eventBridge, BRIDGE_EVENTS } from '../phaser/EventBridge';
import { SPONSOR_CONFIG } from '../data/sponsorConfig';
import { getSponsorSpritePath } from '../data/spriteAssetPaths';

type CapstonePhase = 'dim' | 'beat' | 'fanfare' | 'npc' | 'line2' | 'cert';

export interface CertificateOverlayProps {
  /** Called when the player chooses to return to the start menu. */
  onReturn: () => void;
}

export function CertificateOverlay({ onReturn }: CertificateOverlayProps): JSX.Element {
  const [phase, setPhase] = useState<CapstonePhase>('dim');
  const [copied, setCopied] = useState(false);
  const fanfareFiredRef = useRef(false);

  // ── Sequence pacing ───────────────────────────────────────────
  useEffect(() => {
    if (phase === 'dim') {
      const t = setTimeout(() => setPhase('beat'), 400);
      return () => clearTimeout(t);
    }
    if (phase === 'beat') {
      // The 500ms anticipatory beat (Commandment 2) — pure silence + black.
      const t = setTimeout(() => setPhase('fanfare'), 500);
      return () => clearTimeout(t);
    }
    if (phase === 'fanfare') {
      if (!fanfareFiredRef.current) {
        fanfareFiredRef.current = true;
        eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_fanfare', volume: 0.9 });
      }
      const t = setTimeout(() => setPhase('npc'), 600);
      return () => clearTimeout(t);
    }
  }, [phase]);

  // ── Advance dialogue / cert reveal ────────────────────────────
  const advance = useCallback(() => {
    if (phase === 'npc') {
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_interact', volume: 0.4 });
      setPhase('line2');
    } else if (phase === 'line2') {
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_interact', volume: 0.4 });
      setPhase('cert');
    }
  }, [phase]);

  // ── Keyboard ───────────────────────────────────────────────────
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        onReturn();
        return;
      }
      if (e.key === ' ' || e.key === 'Enter') {
        e.preventDefault();
        advance();
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [advance, onReturn]);

  // ── Copy code ──────────────────────────────────────────────────
  const handleCopy = useCallback(async () => {
    const code = SPONSOR_CONFIG.code;
    let success = false;
    try {
      if (navigator.clipboard && window.isSecureContext) {
        await navigator.clipboard.writeText(code);
        success = true;
      } else {
        const textarea = document.createElement('textarea');
        textarea.value = code;
        textarea.style.position = 'fixed';
        textarea.style.opacity = '0';
        document.body.appendChild(textarea);
        textarea.select();
        success = document.execCommand('copy');
        document.body.removeChild(textarea);
      }
    } catch { success = false; }
    if (success) {
      setCopied(true);
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_sorter_correct', volume: 0.6 });
      setTimeout(() => setCopied(false), 1200);
    }
  }, []);

  // ── Render: dim/beat/fanfare are the lead-in; npc onward is content ─
  const dimOpacity = phase === 'dim' ? 0 : 1;  // CSS animates the climb
  const showNpc   = phase === 'npc' || phase === 'line2' || phase === 'cert';
  const showLine1 = showNpc;
  const showLine2 = phase === 'line2' || phase === 'cert';
  const showCert  = phase === 'cert';

  return (
    <div
      className="absolute inset-0 z-[200] flex items-center justify-center"
      onClick={advance}
      data-testid="certificate-overlay"
    >
      {/* Black dim — climbs from 0 → 0.92 over 400ms in the 'dim' phase */}
      <div
        className="absolute inset-0 transition-opacity duration-[400ms] ease-out"
        style={{ backgroundColor: 'rgba(0,0,0,0.92)', opacity: dimOpacity }}
      />

      {/* Fanfare flash — pulse during the 'fanfare' phase */}
      {phase === 'fanfare' && (
        <div
          className="absolute inset-0 pointer-events-none"
          style={{
            background: 'radial-gradient(ellipse at center, rgba(255,220,80,0.35) 0%, transparent 70%)',
            animation: 'cert-fanfare-flash 600ms ease-out forwards',
          }}
        />
      )}

      {/* Inline keyframes — keeps Phase 21 self-contained without touching index.css */}
      <style>{`
        @keyframes cert-fanfare-flash {
          0%   { opacity: 0; }
          30%  { opacity: 1; }
          100% { opacity: 0; }
        }
        @keyframes cert-card-in {
          0%   { opacity: 0; transform: scale(0.6); }
          70%  { opacity: 1; transform: scale(1.06); }
          100% { opacity: 1; transform: scale(1); }
        }
        @keyframes cert-npc-in {
          0%   { opacity: 0; transform: translateX(-32px); }
          100% { opacity: 1; transform: translateX(0); }
        }
        @keyframes cert-code-reveal {
          0%   { opacity: 0; letter-spacing: 0.6em; }
          100% { opacity: 1; letter-spacing: 0.15em; }
        }
      `}</style>

      {/* Content stage — only renders from 'npc' onward */}
      {showNpc && (
        <div
          className="relative z-10 flex items-center gap-8 px-6"
          style={{ fontFamily: '"Press Start 2P", monospace' }}
        >
          {/* NPC sprite + dialogue column */}
          <div
            className="flex flex-col items-center gap-3"
            style={{ animation: 'cert-npc-in 500ms ease-out forwards' }}
          >
            <div
              data-testid="cert-npc-sprite"
              style={{
                width: 128,
                height: 128,
                backgroundImage: `url(${getSponsorSpritePath(SPONSOR_CONFIG.character_sprite)})`,
                backgroundSize: '384px 512px',  // 96×128 spritesheet upscaled 4×
                backgroundPosition: '0 0',      // frame 0 (idle-down)
                backgroundRepeat: 'no-repeat',
                imageRendering: 'pixelated',
                filter: 'drop-shadow(0 0 12px rgba(255, 220, 80, 0.5))',
              }}
            />
            {/* Dialogue box — JRPG speech bubble */}
            <div
              className="border-4 border-[#FFD23F] bg-[#1a1a2e]/95 px-5 py-4 max-w-[420px] shadow-[6px_6px_0px_0px_rgba(0,0,0,1)]"
              data-testid="cert-dialogue"
            >
              {showLine1 && (
                <p className="text-[10px] text-white leading-relaxed">
                  {SPONSOR_CONFIG.two_dialogue_lines[0]}
                </p>
              )}
              {showLine2 && (
                <p className="text-[10px] text-white leading-relaxed mt-3" data-testid="cert-line-2">
                  {SPONSOR_CONFIG.two_dialogue_lines[1]}
                </p>
              )}
              {!showCert && (
                <p className="text-[7px] text-[#00d4aa] mt-3 tracking-widest">
                  ▼ CLICK / SPACE
                </p>
              )}
            </div>
          </div>

          {/* Certificate card — only in 'cert' phase */}
          {showCert && (
            <div
              className="border-4 border-[#FFD23F] bg-[#1a1a2e] px-8 py-6 shadow-[10px_10px_0px_0px_rgba(0,0,0,1)] flex flex-col items-center gap-4"
              style={{
                animation: 'cert-card-in 600ms cubic-bezier(0.34, 1.56, 0.64, 1) forwards',
                minWidth: 360,
              }}
              data-testid="cert-card"
            >
              <p className="text-[8px] tracking-[0.4em] text-[#FFD23F]">
                CERTIFICATE OF SURVIVAL
              </p>
              <h2
                className="text-[16px] text-white text-center"
                data-testid="cert-sponsor-name"
              >
                {SPONSOR_CONFIG.name}
              </h2>
              <div className="border-2 border-[#FFD23F] bg-black/60 px-6 py-3">
                <p
                  className="text-[14px] text-[#FFD23F]"
                  style={{
                    fontFamily: '"Press Start 2P", monospace',
                    animation: 'cert-code-reveal 700ms ease-out forwards',
                  }}
                  data-testid="cert-code"
                >
                  {SPONSOR_CONFIG.code}
                </p>
              </div>
              <div className="flex gap-3 mt-2">
                <button
                  onClick={(e) => { e.stopPropagation(); handleCopy(); }}
                  data-testid="cert-copy"
                  className={`text-[9px] px-4 py-2 border-4 border-black shadow-[3px_3px_0px_0px_rgba(0,0,0,1)] transition-colors ${
                    copied
                      ? 'bg-[#2ECC71] text-black'
                      : 'bg-[#FFD23F] hover:bg-[#FFC91F] text-black'
                  }`}
                >
                  {copied ? 'COPIED ✓' : 'COPY CODE'}
                </button>
                <button
                  onClick={(e) => { e.stopPropagation(); onReturn(); }}
                  data-testid="cert-return"
                  className="text-[9px] px-4 py-2 border-4 border-black shadow-[3px_3px_0px_0px_rgba(0,0,0,1)] bg-gray-600 hover:bg-gray-500 text-white"
                >
                  RETURN TO MENU
                </button>
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```
