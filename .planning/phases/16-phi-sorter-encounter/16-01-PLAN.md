```typescript
// client/src/data/sorterData.ts — required exports

export type SorterIdentifierType =
  | 'name' | 'geographic' | 'date' | 'phone' | 'fax' | 'email'
  | 'ssn' | 'mrn' | 'plan_id' | 'account' | 'license' | 'vehicle'
  | 'device_serial' | 'url' | 'ip_address' | 'biometric' | 'photo' | 'other';

export type SorterItem = {
  id: string;
  label: string;             // What the player sees on the card
  category: 'phi' | 'not_phi';
  identifierType?: SorterIdentifierType;  // Required when category === 'phi'
  explanation: string;       // Shown on incorrect drop — explains the rule
};

export type SorterDocumentSet = {
  id: string;                // 'phi-sorter-set-1', 'phi-sorter-set-2', 'phi-sorter-set-3'
  act: 1 | 2 | 3;
  triggerLocation: 'reception' | 'lab' | 'medical_records';
  npcId: string;             // For NPC-attributed framing in context card
  contextCard: { title: string; body: string };  // SorterContextCard props
  items: SorterItem[];
  passingAccuracy: number;   // 0.0–1.0; 0.7 default
  takeaways: [string, string]; // EncounterDebrief takeaways (1-2 short HIPAA learnings)
};

export const SORTER_DOCUMENT_SETS: Record<string, SorterDocumentSet>;
export function getSorterDocumentSet(id: string): SorterDocumentSet | undefined;
```

The `takeaways` field is consumed by Plan 04's debrief (rendered via SorterTakeawaysPanel sibling). The `passingAccuracy` is consumed by Plan 04's `handleSorterComplete` to determine 'victory'/'defeat' outcome.
</interfaces>

<hipaa_18_identifiers>
<!-- 45 CFR §164.514(b)(2) Safe Harbor list — Plan 01 items must reflect these accurately. -->
1. Names
2. Geographic data smaller than state (street, city, county, zip — but 3-digit ZIP prefix with >20K population is NOT identifier)
3. Dates more specific than year (DOB, admission, discharge, death, ages 90+)
4. Phone numbers
5. Fax numbers
6. Email addresses
7. Social Security Numbers
8. Medical record numbers
9. Health plan beneficiary numbers
10. Account numbers
11. Certificate/license numbers
12. Vehicle identifiers (incl. plates)
13. Device identifiers/serials
14. Web URLs
15. IP addresses
16. Biometric identifiers
17. Full-face photos
18. Any other unique code

NOT on the list (legitimate "Not PHI" items): organizational addresses (hospital street), aggregate data, year-only dates, 3-digit ZIP prefixes (where >20K), procedure codes without patient connection, environmental measurements (room temperature).
</hipaa_18_identifiers>
