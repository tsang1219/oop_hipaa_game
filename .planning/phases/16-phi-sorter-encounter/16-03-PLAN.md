```tsx
import { useEffect } from 'react';

export type SorterContextCardProps = {
  title: string;
  body: string;
  onConfirm: () => void;
};

export function SorterContextCard({ title, body, onConfirm }: SorterContextCardProps) {
  // Enter/Space confirms — keyboard-first, no mouse required
  useEffect(() => {
    const handleKey = (e: KeyboardEvent) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        onConfirm();
      }
    };
    window.addEventListener('keydown', handleKey);
    return () => window.removeEventListener('keydown', handleKey);
  }, [onConfirm]);

  return (
    <div className="fixed inset-0 z-40 flex items-center justify-center bg-black/70" data-testid="sorter-context-card">
      <div className="max-w-xl w-[90%] bg-[#1a2a3e] border-4 border-[#4FB3D9] shadow-[8px_8px_0px_0px_rgba(0,0,0,1)] p-6">
        <div className="text-[#4FB3D9] text-[10px] font-['Press_Start_2P'] mb-3">{title}</div>
        <div className="text-white text-[10px] leading-relaxed font-['Press_Start_2P'] mb-6 whitespace-pre-wrap">{body}</div>
        <button
          onClick={onConfirm}
          className="bg-[#4FB3D9] hover:bg-[#7ECDE8] text-[#1a1a2e] font-['Press_Start_2P'] text-[10px] px-6 py-3 border-4 border-black"
          data-testid="button-sorter-context-confirm"
```

```tsx
export type BucketZoneProps = {
  bucketType: 'phi' | 'not_phi';
  isHovered: boolean;        // Mouse drag-over OR keyboard ←/→ highlight
  feedbackState: 'idle' | 'correct' | 'incorrect';
  onDrop: (bucketType: 'phi' | 'not_phi') => void;
};

export function BucketZone({ bucketType, isHovered, feedbackState, onDrop }: BucketZoneProps) {
  const isPHI = bucketType === 'phi';
  const baseColor = isPHI ? 'border-[#FF6B9D]' : 'border-[#2ECC71]';
  const label = isPHI ? 'PHI' : 'NOT PHI';
  const sublabel = isPHI ? 'Protected — Redact' : 'Safe to Share';

  const animationClass =
    feedbackState === 'correct' ? 'animate-[flash-green_0.4s_ease-out]' :
    feedbackState === 'incorrect' ? 'animate-[shake-red_0.5s_ease-out]' :
    '';

  return (
    <div
      onDragOver={e => e.preventDefault()}  // REQUIRED for HTML5 drop targets
      onDrop={e => { e.preventDefault(); onDrop(bucketType); }}
      className={[
        'flex flex-col items-center justify-center',
        'min-h-[180px] flex-1 mx-2 p-6 border-4',
        baseColor,
        'bg-[#1a1a2e]/80',
        isHovered ? 'ring-4 ring-white/70 scale-[1.02]' : '',
        animationClass,
        'transition-transform duration-150',
      ].join(' ')}
      data-testid={`bucket-${bucketType}`}
    >
      <div className={`font-['Press_Start_2P'] text-[14px] mb-2 ${isPHI ? 'text-[#FF6B9D]' : 'text-[#2ECC71]'}`}>{label}</div>
      <div className="font-['Press_Start_2P'] text-[8px] text-white/70">{sublabel}</div>

        {isPHI ? 'PRESS ←' : 'PRESS →'}
      </div>

}
```

```tsx
export function PHISorterOverlay({ documentSetId, encounterId, onComplete }: PHISorterOverlayProps) {
  const docSet = useMemo(() => getSorterDocumentSet(documentSetId), [documentSetId]);

  // Error fallback — graceful exit on bad documentSetId
  if (!docSet) {
    return (
      <div className="fixed inset-0 z-40 flex items-center justify-center bg-black/80">
        <div className="bg-[#2a2a3e] border-4 border-red-500 p-6 max-w-sm">
          <p className="text-white font-['Press_Start_2P'] text-[10px] mb-4">
            Sorter content not found: {documentSetId}
          </p>
          <button
            onClick={() => onComplete({ encounterId, correctCount: 0, totalCount: 0, scoreContribution: 0 })}
            className="bg-red-500 text-white font-['Press_Start_2P'] text-[10px] px-4 py-2 border-4 border-black"
          >
            DISMISS
          </button>
        </div>
      </div>
    );
  }

  const [phase, setPhase] = useState<SorterPhase>('preCard');
  const [remainingItems, setRemainingItems] = useState<SorterItemData[]>(docSet.items);
  const [selectedItemIdx, setSelectedItemIdx] = useState(0);
  const [hoveredBucket, setHoveredBucket] = useState<'phi' | 'not_phi' | null>(null);
  const [draggingItemId, setDraggingItemId] = useState<string | null>(null);
  const [correctCount, setCorrectCount] = useState(0);
  const [bucketFeedback, setBucketFeedback] = useState<{ phi: 'idle'|'correct'|'incorrect'; not_phi: 'idle'|'correct'|'incorrect' }>({ phi: 'idle', not_phi: 'idle' });
  const [wrongFeedback, setWrongFeedback] = useState<{ label: string; explanation: string } | null>(null);

  const totalCount = docSet.items.length;

  // Drop handler — used by both mouse drop and keyboard Enter/Space
  const handleDrop = useCallback((itemId: string, bucket: 'phi' | 'not_phi') => {
    const item = remainingItems.find(i => i.id === itemId);
    if (!item) return;

    const isCorrect = item.category === bucket;

    if (isCorrect) {
      setCorrectCount(c => c + 1);
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_sorter_correct', volume: 0.7 });
      setBucketFeedback(f => ({ ...f, [bucket]: 'correct' }));
      setTimeout(() => setBucketFeedback(f => ({ ...f, [bucket]: 'idle' })), 400);
    } else {
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_sorter_wrong', volume: 0.7 });
      setBucketFeedback(f => ({ ...f, [bucket]: 'incorrect' }));
      setTimeout(() => setBucketFeedback(f => ({ ...f, [bucket]: 'idle' })), 500);
      // Educational feedback panel — Commandment 4: teach why
      setWrongFeedback({ label: item.label, explanation: item.explanation });
      setTimeout(() => setWrongFeedback(null), 3500);
    }

    // Remove item; clamp selectedItemIdx
    setRemainingItems(prev => {
      const next = prev.filter(i => i.id !== itemId);
      setSelectedItemIdx(idx => Math.max(0, Math.min(idx, next.length - 1)));
      return next;
    });
    setHoveredBucket(null);
    setDraggingItemId(null);
  }, [remainingItems]);

  // Completion: when sorting and remaining empty → anticipation beat → onComplete
  useEffect(() => {
    if (phase !== 'sorting') return;
    if (remainingItems.length > 0) return;
    setPhase('completing');
    const t = setTimeout(() => {
      eventBridge.emit(BRIDGE_EVENTS.REACT_PLAY_SFX, { key: 'sfx_fanfare', volume: 0.7 });
      const scoreContribution = Math.round((correctCount / totalCount) * 12);
      onComplete({ encounterId, correctCount, totalCount, scoreContribution });
    }, 600);  // Anticipation beat — Commandment 2
    return () => clearTimeout(t);
  }, [phase, remainingItems.length, correctCount, totalCount, encounterId, onComplete]);

  // Keyboard handler — only active in 'sorting' phase
  useEffect(() => {
    if (phase !== 'sorting') return;
    const handleKey = (e: KeyboardEvent) => {
      if (remainingItems.length === 0) return;
      if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
        e.preventDefault();
        const delta = e.key === 'ArrowDown' ? 1 : -1;
        setSelectedItemIdx(idx => (idx + delta + remainingItems.length) % remainingItems.length);
      }
      if (e.key === 'ArrowLeft') { e.preventDefault(); setHoveredBucket('not_phi'); }
      if (e.key === 'ArrowRight') { e.preventDefault(); setHoveredBucket('phi'); }
      if ((e.key === 'Enter' || e.key === ' ') && hoveredBucket) {
        e.preventDefault();
        const item = remainingItems[selectedItemIdx];
        if (item) handleDrop(item.id, hoveredBucket);
      }
    };
    window.addEventListener('keydown', handleKey);
    return () => window.removeEventListener('keydown', handleKey);
  }, [phase, remainingItems, selectedItemIdx, hoveredBucket, handleDrop]);

  // Pre-card phase
  if (phase === 'preCard') {
    return <SorterContextCard
      title={docSet.contextCard.title}
      body={docSet.contextCard.body}
      onConfirm={() => setPhase('sorting')}
    />;
  }

  // Sorting phase render — items pile + buckets + wrong-feedback toast
  return (
    <div className="fixed inset-0 z-40 bg-black/85 flex flex-col items-stretch justify-center p-6" data-testid="phi-sorter-overlay">
      <div className="text-center font-['Press_Start_2P'] text-[10px] text-white mb-4" data-testid="text-sorter-progress">
        {totalCount - remainingItems.length} / {totalCount} sorted · {correctCount} correct
      </div>

      <div className="flex-1 flex flex-row items-stretch justify-center gap-4 max-w-5xl mx-auto w-full">
        {/* Left bucket: NOT PHI */}
        <div className="flex-1 flex items-center">
          <BucketZone
            bucketType="not_phi"
            isHovered={hoveredBucket === 'not_phi' || draggingItemId !== null}
            feedbackState={bucketFeedback.not_phi}
            onDrop={(bt) => draggingItemId &amp;&amp; handleDrop(draggingItemId, bt)}
          />
        </div>

        {/* Center: items pile */}
        <div className="flex-1 max-h-[60vh] overflow-y-auto bg-[#1a1a2e]/60 border-4 border-[#4FB3D9] p-3" data-testid="sorter-items-pile">
          {remainingItems.map((item, idx) => (
            <SorterItem
              key={item.id}
              item={item}
              isSelected={idx === selectedItemIdx}
              isDragging={draggingItemId === item.id}
              onDragStart={(id) => setDraggingItemId(id)}
              onDragEnd={() => setDraggingItemId(null)}
            />
          ))}
        </div>

        {/* Right bucket: PHI */}
        <div className="flex-1 flex items-center">
          <BucketZone
            bucketType="phi"
            isHovered={hoveredBucket === 'phi' || draggingItemId !== null}
            feedbackState={bucketFeedback.phi}
            onDrop={(bt) => draggingItemId &amp;&amp; handleDrop(draggingItemId, bt)}
          />
        </div>
      </div>

      <div className="text-center font-['Press_Start_2P'] text-[8px] text-white/60 mt-4">
        DRAG ITEMS · OR ↑↓ TO SELECT · ←→ FOR BUCKET · ENTER TO COMMIT
      </div>

      {/* Wrong-answer educational toast */}
      {wrongFeedback &amp;&amp; (
        <div className="fixed bottom-6 left-1/2 -translate-x-1/2 max-w-md bg-[#3a1a2a] border-4 border-[#FF6B9D] p-4 z-50" data-testid="sorter-wrong-feedback">
          <div className="text-[#FF6B9D] font-['Press_Start_2P'] text-[8px] mb-1">NOT QUITE — {wrongFeedback.label}</div>
          <div className="text-white font-['Press_Start_2P'] text-[8px] leading-relaxed">{wrongFeedback.explanation}</div>
        </div>
      )}
    </div>
  );
}
```
